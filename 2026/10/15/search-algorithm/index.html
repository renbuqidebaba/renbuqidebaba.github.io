搜索算法（DFS / BFS / 回溯 / 拓扑排序）

搜索是算法题里覆盖面最广的一类。这一篇按题型整理了我做过的搜索题目，主要四类：

  • DFS 暴搜 / 洪水填充——岛屿数量、最大面积、省份数量、水流问题
  • BFS 最短路——二进制矩阵最短路、最短的桥、腐烂橘子
  • 回溯——全排列、组合、子集、括号生成、N 皇后、分割回文串
  • 拓扑排序——课程表（三色标记判环）

回溯部分我单独标了「重复选数」和「不重复选数」的区别，这是写回溯时最容易搞混的地方。

  1. 深度优先搜索（depth-first search，DFS）在搜索到一个新的节点时，立即对该新节点进行遍历；因此遍历需要用先入后出的栈（stack）来实现，也可以通过与栈等价的递归来实现。

判断环的办法：深度优先搜索也可以用来检测环路：记录每个遍历过的节点的父节点，若一个节点被再次遍历且父节点不同，则说明有环。我们也可以用之后会讲到的拓扑排序判断是否有环路，若最后存在入度不为零的点，则说明有环。

有时我们可能会需要对已经搜索过的节点进行标记，以防止在遍历时重复搜索某个节点，这种做法叫做状态记录或记忆化（memoization）。

2.回溯法（backtracking）是优先搜索的一种特殊情况，又称为试探法，常用于需要记录节点状态的深度优先搜索。通常来说，排列、组合、选择类问题使用回溯法比较方便。在搜索到某一节点的时候，如果我们发现目前的节点（及其子节点）并不是需求目标时，我们回退到原来的节点继续搜索，并且把在目前节点修改的状态还原。这样的好处是我们可以始终只对图的总状态进行修改，而非每次遍历时新建一个图来储存状态。在具体的写法上，它与普通的深度优先搜索一样，都有[修改当前节点状态]->[递归子节点] 的步骤，只是多了回溯的步骤，变成了[修改当前节点状态]->[递归子节点]->[回改当前节点状态]。

可以记住两个小诀窍，一是按引用传状态，二是所有的状态修改在递归完成后回改。回溯法修改一般有两种情况，一种是修改最后一位输出，比如排列组合；一种是修改访问标记，比如矩阵里搜字符串。

  1. 广度优先搜索（breadth-first search，BFS）不同于深度优先搜索，它是一层层进行遍历的，因此需要用先入先出的队列(queue) 而非先入后出的栈(stack) 进行遍历。由于是按层次进行遍历，广度优先搜索时按照“广”的方向进行遍历的，也常常用来处理最短路径等问题。

深度优先搜索和广度优先搜索都可以处理可达性问题，即从一个节点开始是否能达到另一个节点。因为深度优先搜索可以利用递归快速实现，很多人会习惯使用深度优先搜索刷此类题目。

Leetcode 695 岛屿最大的面积 mid dfs暴搜

判断边界以及递归的方向，下面是递归的版本。

class Solution {
public:
    int dfs(vector<vector<int>>& grid, int i, int j) {
        if (i < 0 || i >= grid.size() || j < 0 || j >= grid[0].size() || grid[i][j] == 0) return 0;
        grid[i][j] = 0;
        return (1 + dfs(grid, i + 1, j) + dfs(grid, i - 1, j) + dfs(grid, i, j + 1) + dfs(grid, i, j - 1));
    }
    int maxAreaOfIsland(vector<vector<int>>& grid) {
        int maxArea = 0;
        for (int i = 0; i < grid.size(); i++) {
            for (int j = 0; j < grid[0].size(); j++) {
                if (grid[i][j] == 1) {
                    maxArea = max(maxArea, dfs(grid, i ,j));
                }
            }
        }
        return maxArea;
    }
};

Leetcode 547 省份的数量 mid (并查集or dfs邻接表）

第一种思路，用并查集的方法来解决节点连接类的问题，核心是find函数，以及如何知道并查集中不同祖先的个数：设定一个计数器！

class Solution {
public:
    int p[210];
    int count;
    int findCircleNum(vector<vector<int>>& isConnected) {
        init();
        count = isConnected.size();
        for (int i = 0 ; i < isConnected.size(); i++) {
            for (int j = 0; j < isConnected[0].size(); j++) {
                if (isConnected[i][j] && i != j) {
                    if (find(i) != find(j)) {
                        p[find(i)] = find(j);
                        count--;
                    }
                }
            }
        }
        return count;
    }
    void init () {
        for (int i = 1; i < 210; i++) {
            p[i] = i;
        }
    }
    int find(int x) {
        if (p[x] != x) p[x] = find(p[x]);
        return p[x];
    }
};

第二种思路：dfs暴搜

这个图结构有点意思，以前的图结构都是一个点是一个节点并且向四个方向去dfs。但是这个图是以邻接表的形式展示出来，用邻接表的方式来进行搜索。（对于邻接表的遍历思路还是不完善，但是对于四个方向的节点遍历已经回忆的很完善了）。 并且这个题目使用dfs是最快的。

class Solution {
public:
    int findCircleNum(vector<vector<int>>& isConnected) {
        int n = isConnected.size();
        vector<bool> isVisited(n, false);
        int ans = 0;
        for (int i = 0; i < n; i++) {
            if (!isVisited[i]) {
                dfs(isConnected, isVisited, i);
                ans++;
            }
        }
        return ans;
    }
    void dfs(vector<vector<int>>& isConnected, vector<bool>& isVisited, int i) {
        isVisited[i] = true;
        for (int j = 0; j < isConnected.size(); j++) {
            if (isConnected[i][j] && !isVisited[j]) {
                dfs(isConnected, isVisited, j);
            }
        }
    }
};

Leetcode hot 46 全排列 mid 回溯

class Solution {
public:
    void dfs(vector<int>& nums, int u, vector<vector<int>>& pre) {
        if (u == nums.size() - 1) {
            pre.push_back(nums);
        }
        for (int i = u; i <nums.size(); i++) {
            swap(nums[i], nums[u]); // 修改当前节点的状态
            dfs(nums, u + 1, pre);
            swap(nums[i], nums[u]);
        }
    }
    vector<vector<int>> permute(vector<int>& nums) {
        vector<vector<int>> pre;
        dfs(nums, 0, pre);
        return pre;
    }
};

Leetcode 77 组合 mid 回溯

class Solution {
public:
    void dfs(vector<vector<int>>& ans, vector<int>& temp, int n, int k, int u) {
        if (temp.size() == k) {
            ans.push_back(temp);
            return;
        }
        for (int i = u; i <= n; i++) {
            temp.push_back(i);
            dfs(ans, temp, n, k, i + 1);
            temp.pop_back();
        }
    }
    vector<vector<int>> combine(int n, int k) {
        vector<vector<int>> ans;
        vector<int> temp;
        dfs(ans, temp, n, k, 1);
        return ans;
    }
};

Leetcode 1091 二进制矩阵中的最短路径 mid bfs

用bfs进行搜索，但是这个题目多了四个方向可以直接搜索。一开始写出来的版本不是最短的路径很奇怪，也就是存在右边和右下都可以走的时候。（这个题目小蓝书上面的答案是有问题的）

class Solution {
public:
    int shortestPathBinaryMatrix(vector<vector<int>>& grid) {
        if (grid[0][0] == 1) return -1;
        queue<pair<int,int>> myqueue;
        int n = grid.size(), m = grid[0].size();
        vector<vector<int>> dist(n, vector<int>(n, INT_MAX));
        myqueue.push({0, 0});
        dist[0][0] = 1;
        while(myqueue.size() > 0) {
            auto [x, y] = myqueue.front();
            myqueue.pop();
            if (x == n - 1 && y == n - 1) return dist[x][y];
            for (int dx = - 1; dx <= 1; dx++) {
                for (int dy = -1; dy <= 1; dy++) {
                    if (dx == 0 && dy == 0) continue;
                    int a = x + dx, b = y + dy;
                    if (a < 0 || a >= n || b < 0 || b >= m || grid[a][b] != 0 || dist[a][b] <= dist[x][y] + 1) continue;
                    dist[a][b] = dist[x][y] + 1;
                    myqueue.push({a,b});
                }
            }
        }
        return -1;
    }
};

Leetcode 934 最短的桥 mid （dfs+bfs）

这个题目是dfs和bfs的综合题目，感觉是两种方法，一种是设定dist[n][n]来计算最小值，一种是两重循环看走了几个层数（第一个统计层数，第二个执行具体的逻辑，一般层数的叠加一般在最后面）

class Solution {
public:
    vector<int> direction {-1, 0, 1, 0, -1};
<![CDATA[    void dfs(queue<pair<int, int>>& q, vector<vector<int>>& grid, int i, int j) {]]>
        int n = grid.size(), m = grid[0].size(); 
        if (i >= n || i < 0 || j >= m || j < 0 || grid[i][j] != 1) return;
        if (grid[i][j] == 1) {
            grid[i][j] = 2;//为第一个图设置不一样的标志
            q.push({i, j});
        }
        dfs(q, grid, i + 1, j);
        dfs(q, grid, i - 1, j);
        dfs(q, grid, i, j + 1);
        dfs(q, grid, i, j - 1);
    }
    int shortestBridge(vector<vector<int>>& grid) {
        int n = grid.size(), m = grid[0].size();
        queue<pair<int, int>> q;
        bool flag = false;
        for (int i = 0; i < n && !flag; i++) {
            for (int j = 0; j < m && !flag; j++) {
                if (grid[i][j] == 1) {
                    dfs(q, grid, i, j);
                    flag = true;
                }
            }
        }
        int ans = 0;
        while(!q.empty()) {
            int q_size = q.size();
            while(q_size--) {
                auto [r, c] = q.front();
                q.pop();
                for (int i = 0; i < 4; i++) {
                    int x = r + direction[i], y = c + direction[i + 1];
                    if (x >= n || x < 0 || y >= m || y < 0 || grid[x][y] == 2 || grid[x][y] == -1) continue;
                    if (grid[x][y] == 1) return ans;
                    // 只剩下0的情况了
                    grid[x][y] = -1;
                    q.push({x, y});
                }
            }
            ans++;
        }
        return ans;
    }
};

Leetcode 257 二叉树的所有路径 easy

dfs搜索，要熟悉对于二叉树的dfs搜索方式！

/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    void dfs(TreeNode* root, string path, vector<string>& ans) {
        if (root == nullptr) return;
        path += to_string(root -> val);
        //如果当前是叶子节点，那么就加入到答案中
        if (root -> left == nullptr && root -> right == nullptr) {
            ans.push_back(path);
        } else {
            //如果不是叶子节点，继续往下递归
            path += "->";
            dfs(root -> left, path, ans);
            dfs(root -> right, path, ans);
        }

    }
    vector<string> binaryTreePaths(TreeNode* root) {
        vector<string> ans;
        dfs(root, "", ans);
        return ans;
    }
};

Leetcode hot 39 组合总和 mid （重复选数的回溯）

找工作之前重新温习，没有看答案自己按照理解ac代码。

这里u在走的时候表示取不取数组中的第u个元素，这和全排列的定义不一样，全排列的u是第u位放什么元素。那么就有两种情况：选当前元素，递归到下一层的时候u不变，因为可以重复选；不选当前元素，递归到下一层的时候，u要加1，表示选下一个元素。

class Solution:
    def combinationSum(self, candidates: List[int], target: int) -> List[List[int]]:
        self.ans = []
        self.temp = []
        self.dfs(0, candidates, target)
        return self.ans
    
    def dfs(self, u, candidates, target):
        n = len(candidates)
        # 说明元素用完了
        if u == n:
            return
        # 说明到了答案
        if target == 0:
            self.ans.append(list(self.temp))
            return
        # 不选当前的元素，就需要走到下一个位置
        self.dfs(u + 1, candidates, target)
        if target >= candidates[u]:
            self.temp.append(candidates[u])
            # 选择了当前元素，不需要走到下一个位置，因为可以重复选
            self.dfs(u, candidates, target - candidates[u])
            self.temp.pop()

Leetcode 22 括号生成 mid 回溯

这个题目的回溯其实自己没有想好，主要是没有想到这种两个括号组合判断方式，跟上一个题目是不一样的。这个一个位置能不能放置括号是根据目前的个数进行判断的！（设定两个参数，一个目前用的括号数，一个是其中左括号数目）

class Solution {
public:
    void dfs(int u , int open, vector<string>& ans, int n, string& temp) {
        if (u == n*2) {
            ans.push_back(temp);
            return;
        }
        if (open < n) {//左边括号数量
            temp.push_back('(');
            dfs(u + 1, open + 1, ans, n, temp);
            temp.pop_back();
        }
        if (u < 2*open) {
            temp.push_back(')');
            dfs(u + 1, open, ans, n, temp);
            temp.pop_back();
        }
    }
    vector<string> generateParenthesis(int n) {
        vector<string> ans;
        string temp;
        dfs(0, 0, ans, n, temp);
        return ans;
    }
};

Leetcode 105 从前序和中序遍历构建二叉树

找到前序和中序遍历的特点就行。然后递归到下一层构建树，树就用后序遍历进行构建。

/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    TreeNode* buildTree(vector<int>& preorder, vector<int>& inorder) {
        if (preorder.size() == 0 || inorder.size() == 0) return nullptr;
        return dfs(preorder, inorder);
    }
    TreeNode* dfs(vector<int>& preorder, vector<int>& inorder) {
        if (preorder.size() == 0) return nullptr;
        int rootv = preorder[0];
        TreeNode* root = new TreeNode(rootv);
        int i; //i是该元素在中序遍历中的位置
        for (i = 0; i < inorder.size(); i++) {
            if (inorder[i] == rootv) break;
        }
        vector<int> leftInorder(inorder.begin(),inorder.begin() + i);
        vector<int> rigthInorder(inorder.begin() + i + 1, inorder.end());
        vector<int> leftPre(preorder.begin() + 1, preorder.begin() + 1 + leftInorder.size());
        vector<int> rightPre(preorder.begin() + 1 + leftInorder.size(), preorder.end());
        root -> left = dfs(leftPre, leftInorder);
        root -> right = dfs(rightPre, rigthInorder);
        return root;
    }
};

Leetcode 131 分割回文串 mid

分割问题的可以将每一个字母之间含有逗号来进行处理，并且进行回溯递归. 枚举逗号相当于是枚举回文子串结束的位置。

class Solution {
public:
    bool isHW(string s, int left, int right) {
        while (left < right) {
            if (s[left++] != s[right--]) return false;
        }
        return true;
    }
    void dfs(vector<vector<string>>& ans, vector<string>& path, int i, int n, int st, string s) {
        if (i == n) {
            ans.push_back(path);
            return;
        }
        //不选择下一个逗号分割，如果是最后一个位置是一定要选的
        if (i < n - 1) {
            dfs(ans, path, i + 1, n, st, s);
        }
        //选择逗号
        if (isHW(s, st, i)) {
            path.push_back(s.substr(st, i - st + 1));
            dfs(ans, path, i + 1, n, i + 1, s);
            path.pop_back();
        }
    }
    vector<vector<string>> partition(string s) {
        int n = s.size();
        vector<vector<string>> ans;
        vector<string> path;
        dfs(ans, path, 0, n, 0, s);
        return ans;
    }
    
};

Leetcode 51 n皇后 hard

其实就是全排列的变体，但是看似多了一个维度，其实不然，n皇后不仅要确保列没有，还要确保两个对角线没有。其余的思路其实跟全排列是一样的。在递归的时候是考虑皇后可以被放在哪一个列上去。所以需要一个vector来记录第几行放在哪一个列的。然后其余的就是限制条件了。需要一个变量u告诉走到了第几行。

Leetcode hot 207 课程表 mid （top排序+三色标记）

top排序或者是使用0X3F的三色标记法！！！节点颜色分为三类 0表示还没有进行处理，1表示正在进行处理， 2表示处理完成了。dfs去搜索判断逻辑和主函数一致，主要还是想清楚搜索的思路。

class Solution {
public:
    bool canFinish(int numCourses, vector<vector<int>>& prerequisites) {
        vector<vector<int>> g(numCourses);
        for (auto& x : prerequisites) {
            g[x[1]].push_back(x[0]);
        }
        vector<int> colos(numCourses);
        for (int i = 0; i < numCourses; i++) {
            if (colos[i] == 0 && dfs(i, colos, g)) return false;
        }
        return true;
    }
    bool dfs(int x, vector<int>& colos, vector<vector<int>>& g) {
        colos[x] = 1; // 表示节点正在访问中
        for (int y : g[x]) {
            if (colos[y] == 1 || colos[y] == 0 && dfs(y, colos, g)) return true;//找到了环
        }
        colos[x] = 2 ;// 2表示访问完毕
        return false;
    }
};

func dfs(u int, colos []int, g [][]int ) bool {
    // 1表示正在被处理， 0表示还没有被处理， 2表示处理完毕
    colos[u] = 1
    for _, v := range g[u] {
        if colos[v] == 1 || colos[v] == 0 && dfs(v, colos, g) {
            return true
        }
    }
    colos[u] = 2 // 邻居访问完毕
    return false
}

Pythonclass Solution:
    def canFinish(self, numCourses: int, prerequisites: List[List[int]]) -> bool:
        grid = [[] for i in range(numCourses)]
        # 三色标记，0表示未访问，1表示正在访问，2表示已经访问完成
        colos = [0]*numCourses
        for x in prerequisites:
            grid[x[1]].append(x[0])
            
        for i in range(numCourses):
            # 和dfs中条件一致也是可以ac的，但是两者放回置不一样，dfs是找有没有环的
            if colos[i] == 1 or (colos[i] == 0 and self.dfs(i, grid, colos)):
                return False
        return True
        
    # dfs找为true就是找到了环，为false就是没有找到环
    def dfs(self, u, grid, colos):
        colos[u] = 1
        for i in grid[u]:
            if colos[i] == 1 or (colos[i] == 0 and self.dfs(i, grid, colos)):
                return True
        colos[u] = 2
        return False

Leetcode 200 岛屿数量 mid

很经典的dfs搜索题目，因为要区分岛屿的数量。

其实考察就是连通性问题，那么肯定首选dfs，然后再记录dfs多少次。

import sys
def dfs(grid, n, m, i, j):
    if  i >=  n or i < 0 or j >= m  or j < 0 or grid[i][j] == "0":
        return
    grid[i][j] = "0"
    dfs(grid, n, m, i + 1, j)
    dfs(grid, n, m, i, j + 1)
    dfs(grid, m, m, i - 1, j)
    dfs(grid, n, m, i, j - 1)

if __name__ == "__main__":
    input = sys.stdin.read

    n, m = map(int, datas[0].split())
    grid = []
    for i in range(1, n + 1):
        grid.append(list(datas[i]))
    counter = 0
    for i in range(n):
        for j in range(m):
            if grid[i][j] == "1":
                dfs(grid, n, m, i, j)
                counter += 1
    print(counter)

Codefun 4020 腐烂橘子 bfs

import sys
from collections import deque

class Solution:
    def bfs(self, grid):
        n, m = len(grid), len(grid[0])
        queue = deque()
        fresh_count = 0

        for i in range(n):
            for j in range(m):
                if grid[i][j] == "2":
                    queue.append((i, j, 0))
                elif grid[i][j] == "1":
                    fresh_count += 1
        
        if fresh_count == 0:
            return 0
        
        diret = [(0, 1), (0, -1), (1, 0), (-1, 0)]
        minutes = 0

        while queue:
            x, y, minutes = queue.popleft()
            for dx, dy in diret:
                nx, ny = x + dx, y + dy
                if nx >= 0 and nx < n and ny >= 0 and ny < m and grid[nx][ny] == "1":
                    grid[nx][ny] = "2"
                    queue.append((nx, ny, minutes + 1))
                    fresh_count -= 1
        
        return -1 if fresh_count > 0 else minutes
    

if __name__ == "__main__":

    datas = input().split("\n")
    n, m = map(int, datas[0].split())
    graph = [datas[i].split() for i in range(1, n + 1)]
    solution = Solution()
    print(solution.bfs(graph))

字节腾讯常见面试题 从数组中构建最大不超过n的数

dfs搜索+贪心。

import sys

def dfs(u, nums, n_str, ans, flag):
    if u == len(n_str):
        return
    # 开始选择元素
    temp = int(n_str[u])
    idx = len(nums) - 1
    # true表示还在选择元素，false表示前面的一定小于，后面直接选最大的
    if flag:
        while nums[idx] > temp:
            idx -= 1
        # 如果是相等，说明后面的元素还需要选择
        if nums[idx] == temp:
            ans.append(nums[idx])
            dfs(u + 1, nums, n_str, ans, True)
        # 如果是比当前的位数小，后面全部选最大的就行，直接返回
        if nums[idx] < temp:
            ans.append(nums[idx])
            dfs(u + 1, nums, n_str, ans, False)
    else:
        ans.append(nums[idx])
        dfs(u + 1, nums, n_str, ans, False)

def maxnums(nums, n):
    n_str = str(n)
    nums.sort()
    ans = []
    dfs(0, nums, n_str, ans, True)
    return ans

if __name__ == "__main__":
    n = 239
    nums = [2, 1, 9]
    ans = maxnums(nums, n)
    print(ans)
