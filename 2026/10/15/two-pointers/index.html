双指针

双指针是用两个下标在数组/链表上单向扫描，把 O(n²) 的暴力降到 O(n)。常见形态有三种：左右指针（相向）、快慢指针（同向不同速）、滑动窗口（维护一段区间）。

下面是按题目整理的笔记。

算法的思路：有两个指针来进行索引。

若两个指针指向同一数组，遍历方向相同且不会相交，则也称为滑动窗口（两个指针包围的区域即为当前的窗口），经常用于区间搜索

若两个指针指向同一数组，但是遍历方向相反，则可以用来进行搜索，待搜索的数组往往是排好序的

快慢指针问题：是否存在环？快慢指针（一个指针走两步，另外一个走一步），如果是相遇的说明有环。第二个如何判断链表开头的位置？相遇后把快指针放于首位，改为移动一步，再次相遇时就是链表的入口。对于某些只需要判断是否存在环路的题目，也可以通过建造哈希表来查重。

Leetcode 两数之和有序 easy

双指针的基本思路，前提是有序。最后直接返回vector{i + 1, j + 1}可以达到最快的速度.

class Solution {
public:
    vector<int> twoSum(vector<int>& numbers, int target) {
        vector<int> ans;
        int n = numbers.size(),i = 0, j = n-1;
        while(i < n && j >= 0) {
            if(numbers[i] + numbers[j] == target) {
                ans.push_back(i+1);
                ans.push_back(j+1);
                break;
            }
            if(numbers[i] + numbers[j] > target) {
                j--;
            }
            if(numbers[i] + numbers[j] < target) {
                i++;
            }
        }
        return ans;
    }
};

Leetcode 88 合并两个有序数组 mid

从后往前进行操作，这样可以保证num1前面的数据不会被直接覆盖。（反向思考）

class Solution {
public:
    void merge(vector<int>& nums1, int m, vector<int>& nums2, int n) {
        int i = m - 1, j = n - 1, k = m + n - 1;
        while(j >= 0 && i >= 0) {
            nums1[k--] = nums1[i] > nums2[j] ? nums1[i--] : nums2[j--];
        }
        //num2还没有完，这里不用考虑num1没有完的情况，如果num2完了，因为num1比num2大，那么剩下的一定是有序的。
        while(j >= 0){
            nums1[k--] = nums2[j--];
        }
    }
};

Leetcode 142 环形链表 mid

快慢指针的应用，如何判断是否存在环形，具体见上面的双指针讲解。

/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode(int x) : val(x), next(NULL) {}
 * };
 */
class Solution {
public:
    ListNode *detectCycle(ListNode *head) {
        ListNode *first = head, *slow = head;
        bool is_first = true;
        while(first != slow || is_first) {
            if(first == nullptr || first->next == nullptr) {
                return nullptr;
            }
            first = first->next->next;
            slow = slow->next;
            is_first = false;
        }
        first = head;
        while(first != slow) {
            first = first->next;
            slow = slow->next;
        }
        return first;
    }
};

Leetcode 76 最小覆盖子串 hard

这个题目还是没有完全理解，但是基本上还是差不多，还是需要多思考

class Solution {
public:
    string minWindow(string s, string t) {
        vector<bool> vaild(128, false); //表示字符是否在t中存在
        vector<int> freq(128, 0); // 表示t中每个字符在s的滑动窗口中缺少的数量
        for (int i = 0; i < t.size(); i++) {
            vaild[t[i]] = true;
            freq[t[i]]++;
        }
        int count = 0;
        int min_l = -1, mini_length = -1;
        for(int l = 0, r = 0; r < s.size(); r++) {
            if(!vaild[s[r]]) continue;
            if(--freq[s[r]] >= 0) count++;
            while(count == t.size()) {
                if(mini_length == -1 || r - l + 1 < mini_length) {
                    min_l = l;
                    mini_length = r - l + 1;
                }
                if (vaild[s[l]] && ++freq[s[l]] > 0) count--;
                l++;
            }
        }
        return mini_length == -1 ? "" : s.substr(min_l, mini_length);
    }
};

Leetcode 633 平方数之和 mid

思路还是完全掌握了，结题过程中发现了超出int，采用更大的long来就ok，思路是完全没有问题的双指针。

class Solution {
public:
    bool judgeSquareSum(int c) {
        long i = 0, j = sqrt(c);
        while (i <= j) {
            long n = 2*i*j;
            long num = i + j;
            if (num == sqrt(c + n)) return true;
            if (num > sqrt(c + n)) j--;
            if (num < sqrt(c + n)) i++;
        }
        return false;
    }
};

md分析了错误的测试样例终于知道错误的地方了，真的不简单啊我测。要判断全左移或者全右移，不能每一次选择两边的，不然很容易被cuppucu情况吃掉。

class Solution {
public:
    bool validPalindrome(string s) {
        int i = -1, j = s.size();
        bool isfirst = true;
        bool first = true, second = true;
        while(i < j) {
            i++;
            j--;
            if (s[i] != s[j]) {
                //发生转移
                if (isfirst) {
                    if (s[i + 1] == s[j]) {
                        i++;
                        isfirst = false;
                        continue;
                    }
                }
                first = false;
                break;
            }
        }
        i = -1, j = s.size();
        isfirst = true;
        while(i < j) {
            i++;
            j--;
            if (s[i] != s[j]) {
                //发生转移
                if (isfirst) {
                    if (s[i] == s[j - 1]) {
                        j--;
                        isfirst = false;
                        continue;
                    }
                }
                second = false;
                break;
            }
        }
        if (!first && !second) return false;
        return true;
    }
};

Acwing 799 最长连续不重复子序列 easy

滑动窗口+map去重。第一遍写出来大概逻辑基本上没问题，小修改2次就ac

#include<iostream>
#include<cstring>
#include<algorithm>
using namespace std;
const int N = 100010;
int a[N], s[N];
int n;
int main() {
    scanf("%d", &n);
    for (int i = 0 ; i < n; i++) scanf("%d", &a[i]);
    int res = 0;
    for (int i = 0, j = 0 ; j < n; j++) {
        s[a[j]]++;
        while (s[a[j]] > 1 && i <= j) {
            s[a[i]]--;
            i++;
        }
        res = max(res, j - i + 1);
    }
    printf("%d\n", res);
}

Leetcode 3 无重复字符的最长子串 mid

class Solution:
    def lengthOfLongestSubstring(self, s: str) -> int:
        n = len(s)
        mmap = {}
        j, ans = 0, 0
        for i in range(n):
            mmap[s[i]] = mmap.get(s[i], 0) + 1
            while j < i and mmap[s[i]] > 1:
                mmap[s[j]] -= 1
                j += 1
            # 全部移动完了，才更新
            ans = max(ans, i - j + 1)
        return ans

Codefun 三数之和

import sys

class Solution:
    def __init__(self, n, num):
        self.n = n 
        self.num = num
    
    def ThreeNums(self):
        self.num.sort()
        ans = set()
        for i in range(self.n - 1):
            j, k = i + 1, self.n - 1
            while j < k:
                if num[i] + num[j] + num[k] == 0:
                    ans.add((num[i], num[j], num[k]))
                    j += 1
                if num[i] + num[j] + num[k] > 0:
                    k -= 1
                if num[i] + num[j] + num[k] < 0:
                    j += 1
        return ans

Leetcode hot 42 接雨水 hard

双指针的思路去做这个题目,相当于只管较小的一边，因为较大的一边肯定可以兜住，然后一段段累加。维护两个指针和两个最大值。

class Solution:
    def trap(self, height: List[int]) -> int:
        n = len(height)
        left, right = 0, n - 1
        leftMax, rightMax = 0, 0
        ans = 0
        # 双指针的思路去做这个题目,相当于只管较小的一边，因为较大的一边肯定可以兜住，然后一段段累加
        while left < right:
            leftMax = max(leftMax, height[left])
            rightMax = max(rightMax, height[right])
            if height[left] < height[right]:
                ans += (leftMax - height[left])
                left += 1
            else:
                ans += (rightMax - height[right])
                right -= 1
        
        return ans
